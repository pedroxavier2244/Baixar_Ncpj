# E2E QA Test — Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Criar teste ponta a ponta completo (pipeline ETL + API HTTP) com PostgreSQL efêmero via Docker, sem depender da Receita Federal real.

**Architecture:** `run_e2e.bat` sobe container Docker (postgres:15-alpine, porta 5433), aplica o schema via `docker exec psql`, chama `tests/e2e_test.py` que cria fixtures no formato real da RF, roda os 5 steps do pipeline (verify → extract → transform → load → index), valida os dados via SQL direto e depois sobe uvicorn para testar os endpoints HTTP. Container destruído ao final independente de sucesso ou falha.

**Tech Stack:** Python 3.13, psycopg 3, FastAPI, httpx, uvicorn, Docker (postgres:15-alpine), Windows BAT

## Dados de fixture

O pipeline espera arquivos no formato real da RF:
- Encoding: **latin-1**
- Separador: **`;`**
- **Sem header** nas linhas (transform_step.py adiciona os headers)
- Nomes de arquivo internos nos ZIPs no formato `K3241.K03200Y0.D51213.EMPRECSV`

CNPJ de referência para assertions:
- `cnpj_basico = "11111111"`, `cnpj_ordem = "0001"`, `cnpj_dv = "41"`
- `cnpj_completo = "11111111000141"` (14 chars = basico+ordem+dv)

**Step 1: Criar o arquivo**

```bat
@echo off
setlocal

:: ── Config ────────────────────────────────────────────────────────────────
set CONTAINER=cnpj_e2e_test
set PG_PORT=5433
set PG_PASS=test
set POSTGRES_URL=postgresql://postgres:%PG_PASS%@localhost:%PG_PORT%/postgres

set RESULT=0

:: ── Remove container anterior se existir ──────────────────────────────────
docker rm -f %CONTAINER% >nul 2>&1

:: ── Start PostgreSQL ───────────────────────────────────────────────────────
echo [e2e] Subindo PostgreSQL...
docker run -d --name %CONTAINER% ^
    -p %PG_PORT%:5432 ^
    -e POSTGRES_PASSWORD=%PG_PASS% ^
    postgres:15-alpine >nul 2>&1

if %ERRORLEVEL% neq 0 (
    echo [e2e] ERRO: falha ao iniciar container Docker.
    echo [e2e] Docker esta rodando? docker ps
    exit /b 1
)

:: ── Aguardar PostgreSQL estar pronto ──────────────────────────────────────
echo [e2e] Aguardando PostgreSQL...
set attempts=0
:wait_pg
docker exec %CONTAINER% pg_isready -U postgres >nul 2>&1
if %ERRORLEVEL% equ 0 goto pg_ready
set /a attempts+=1
if %attempts% geq 30 (
    echo [e2e] ERRO: PostgreSQL nao iniciou em 30s.
    set RESULT=1
    goto cleanup
)
timeout /t 1 /nobreak >nul
goto wait_pg

:pg_ready
echo [e2e] PostgreSQL pronto.

:: ── Aplicar schema ────────────────────────────────────────────────────────
echo [e2e] Aplicando schema.sql...
docker cp db\schema.sql %CONTAINER%:/tmp/schema.sql
docker exec %CONTAINER% psql -U postgres -f /tmp/schema.sql >nul 2>&1
if %ERRORLEVEL% neq 0 (
    echo [e2e] ERRO: falha ao aplicar schema.sql
    set RESULT=1
    goto cleanup
)
echo [e2e] Schema aplicado.

:: ── Rodar testes ──────────────────────────────────────────────────────────
echo [e2e] Executando e2e_test.py...
python tests\e2e_test.py
set RESULT=%ERRORLEVEL%

:: ── Cleanup ───────────────────────────────────────────────────────────────
:cleanup
echo [e2e] Removendo container...
docker stop %CONTAINER% >nul 2>&1
docker rm   %CONTAINER% >nul 2>&1

echo.
if %RESULT% equ 0 (
    echo [e2e] ============================================
    echo [e2e]  TODOS OS TESTES PASSARAM
    echo [e2e] ============================================
) else (
    echo [e2e] ============================================
    echo [e2e]  TESTES FALHARAM
    echo [e2e] ============================================
)
exit /b %RESULT%
```

        _passed += 1

        _failed += 1

        print(f"  FAIL  {label}{suffix}")

# ═══════════════════════════════════════════════════════════════════════════

def phase2_fixtures() -> None:
    print("\n=== FASE 2: Fixtures ===")
    ZIP_DIR.mkdir(parents=True, exist_ok=True)
    CHECKPOINT_DIR.mkdir(parents=True, exist_ok=True)

    download_results = []
    for zip_name, (inner_name, content) in ZIP_FIXTURES.items():
        zip_path = ZIP_DIR / zip_name
        with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as zf:
            zf.writestr(inner_name, content.encode("latin-1"))
        size = zip_path.stat().st_size
        download_results.append({"name": zip_name, "status": "exists", "bytes": size})

    # Injetar download_manifest.json — faz o verify_step achar os ZIPs
    manifest = {"run_key": RUN_KEY, "results": download_results}
    (CHECKPOINT_DIR / "download_manifest.json").write_text(
        json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    check(f"{len(download_results)} ZIPs criados com nomes internos RF", True)
    check("download_manifest.json injetado no checkpoint_dir", True)

            return conn.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]

            return row[0] if row else None

              count("cnpj.rf_empresas") == 3,

# ═══════════════════════════════════════════════════════════════════════════
# FASE 5 — API HTTP: subir uvicorn e testar endpoints
# ═══════════════════════════════════════════════════════════════════════════

def phase5_api() -> None:
    print("\n=== FASE 5: API HTTP ===")

    env = os.environ.copy()
    env["POSTGRES_URL"] = POSTGRES_URL

    proc = subprocess.Popen(
        [
            sys.executable, "-m", "uvicorn", "api.main:app",
            "--host", "127.0.0.1",
            "--port", str(API_PORT),
            "--log-level", "warning",
        ],
        env=env,
        cwd=str(PROJECT_ROOT),
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL,
    )

    base = f"http://127.0.0.1:{API_PORT}"
    api_up = False
    for _ in range(20):
        time.sleep(1)
        try:
            r = httpx.get(f"{base}/health", timeout=2)
            if r.status_code == 200:
                api_up = True
                break
        except Exception:
            pass

    check("API uvicorn iniciou e respondeu /health", api_up)
    if not api_up:
        proc.terminate()
        return

    try:
        with httpx.Client(base_url=base, timeout=10) as client:

            # ── /health ────────────────────────────────────────────────────
            r = client.get("/health")
            check("GET /health status=200", r.status_code == 200)
            if r.status_code == 200:
                h = r.json()
                check("GET /health retorna status='ok'",
                      h.get("status") == "ok", detail=str(h.get("status")))
                check("GET /health mv_row_count=3",
                      h.get("mv_row_count") == 3,
                      detail=f"encontrado: {h.get('mv_row_count')}")

            # ── /cnpj/{cnpj} CNPJ completo (14 digitos) ───────────────────
            r = client.get(f"/cnpj/{CNPJ_COMPLETO}")
            check(f"GET /cnpj/{CNPJ_COMPLETO} status=200", r.status_code == 200)
            if r.status_code == 200:
                d = r.json()
                check("GET /cnpj retorna razao_social correta",
                      d.get("razao_social") == "EMPRESA TESTE LTDA",
                      detail=d.get("razao_social"))
                check("GET /cnpj retorna uf=SP",
                      d.get("uf") == "SP", detail=d.get("uf"))
                check("GET /cnpj retorna cnpj_completo correto",
                      d.get("cnpj_completo") == CNPJ_COMPLETO,
                      detail=d.get("cnpj_completo"))

            # ── /cnpj/{cnpj} CNPJ basico (8 digitos) ─────────────────────
            r = client.get(f"/cnpj/{CNPJ_BASICO}")
            check(f"GET /cnpj/{CNPJ_BASICO} (basico) status=200",
                  r.status_code == 200)

            # ── /cnpj/{cnpj} validacoes de erro ───────────────────────────
            r = client.get("/cnpj/123")
            check("GET /cnpj/123 (invalido) retorna 400",
                  r.status_code == 400)

            r = client.get("/cnpj/99999999999999")
            check("GET /cnpj/99999999999999 (nao existe) retorna 404",
                  r.status_code == 404)

            # ── /search ───────────────────────────────────────────────────
            r = client.get("/search", params={"razao_social": "TESTE"})
            check("GET /search?razao_social=TESTE status=200",
                  r.status_code == 200)
